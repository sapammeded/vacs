/**
 * VACS V6.4 — FINAL SIMPLE TWO-SHEET BACKEND
 *
 * ONLY TWO SHEETS ARE USED:
 * 1. VEHICLES
 * 2. ACCESS_LOG
 *
 * VEHICLES:
 * Nama | Nomor Kendaraan | Bagian/Divisi | Jenis Kendaraan | Model | Warna Kendaraan
 *
 * ACCESS_LOG:
 * Nama | Nomor Kendaraan | Bagian/Divisi | Jenis Kendaraan | Model | Warna Kendaraan |
 * Jam Masuk | Jam Keluar | Hari/Tanggal | Keterangan
 */

const CONFIG = {
  SPREADSHEET_ID: '1wFnASAShpQgFA4bMZE2PIH5ccKma66otpHNMjLZgyv4',
  TIMEZONE: 'Asia/Jakarta',
  VERSION: '6.5'
};

const VEHICLE_HEADERS = [
  'Nama','Nomor Kendaraan','Bagian/Divisi','Jenis Kendaraan','Model','Warna Kendaraan'
];

const LOG_HEADERS = [
  'Nama','Nomor Kendaraan','Bagian/Divisi','Jenis Kendaraan','Model','Warna Kendaraan',
  'Jam Masuk','Jam Keluar','Hari/Tanggal','Keterangan'
];

function db_(){ return SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID); }

function normalize_(v){
  return String(v == null ? '' : v).toUpperCase().replace(/[^A-Z0-9]/g,'');
}

function now_(){
  const d=new Date();
  return {
    timestamp:Utilities.formatDate(d,CONFIG.TIMEZONE,'yyyy-MM-dd HH:mm:ss'),
    date:Utilities.formatDate(d,CONFIG.TIMEZONE,'dd/MM/yyyy'),
    time:Utilities.formatDate(d,CONFIG.TIMEZONE,'HH:mm:ss'),
    year:Number(Utilities.formatDate(d,CONFIG.TIMEZONE,'yyyy')),
    month:Number(Utilities.formatDate(d,CONFIG.TIMEZONE,'MM')),
    day:Number(Utilities.formatDate(d,CONFIG.TIMEZONE,'dd'))
  };
}

/**
 * IMPORTANT:
 * Every time the API is called, the spreadsheet is automatically normalized.
 * All tabs except VEHICLES and ACCESS_LOG are deleted automatically.
 * No manual cleanup is required.
 */
function ensureDatabase_(){
  const ss=db_();
  let vehicles=ss.getSheetByName('VEHICLES');
  let log=ss.getSheetByName('ACCESS_LOG');

  // If a legacy Sheet1 is the only sheet, reuse it as VEHICLES.
  if(!vehicles && ss.getSheets().length===1){
    const only=ss.getSheets()[0];
    if(only.getName()==='Sheet1') only.setName('VEHICLES');
    vehicles=ss.getSheetByName('VEHICLES');
  }

  if(!vehicles) vehicles=ss.insertSheet('VEHICLES');
  if(!log) log=ss.insertSheet('ACCESS_LOG');

  // Delete every legacy/helper tab automatically.
  ss.getSheets().slice().forEach(sh=>{
    const n=sh.getName();
    if(n!=='VEHICLES' && n!=='ACCESS_LOG'){
      // Spreadsheet must always retain at least one sheet; two required sheets exist.
      ss.deleteSheet(sh);
    }
  });

  // Set exact headers. Do not resize columns or alter the user's widths.
  vehicles.getRange(1,1,1,VEHICLE_HEADERS.length).setValues([VEHICLE_HEADERS]);
  log.getRange(1,1,1,LOG_HEADERS.length).setValues([LOG_HEADERS]);

  vehicles.setFrozenRows(1);
  log.setFrozenRows(1);

  return ss;
}



function setupHeadersOnly_(){
  const ss=db_();
  const v=ss.getSheetByName('VEHICLES');
  const l=ss.getSheetByName('ACCESS_LOG');
  if(!v || !l) return;
  v.getRange(1,1,1,6).setValues([VEHICLE_HEADERS]);
  l.getRange(1,1,1,10).setValues([LOG_HEADERS]);
}
function setupVACS(){
  const ss=ensureDatabase_();
  const v=ss.getSheetByName('VEHICLES');
  const l=ss.getSheetByName('ACCESS_LOG');

  v.getRange(1,1,1,6).setValues([VEHICLE_HEADERS]);
  l.getRange(1,1,1,10).setValues([LOG_HEADERS]);

  v.getRange(1,1,1,6).setFontWeight('bold');
  l.getRange(1,1,1,10).setFontWeight('bold');

  return {
    ok:true,
    message:'VACS siap. Hanya VEHICLES dan ACCESS_LOG yang digunakan.',
    vehiclesHeaders:VEHICLE_HEADERS,
    accessLogHeaders:LOG_HEADERS
  };
}

function doGet(e){
  try{
    ensureDatabase_();
    setupHeadersOnly_();
    const p=(e&&e.parameter)||{};
    const result=route_(String(p.action||'health'),p);
    return response_(p.callback,{ok:true,data:result});
  }catch(err){
    const p=(e&&e.parameter)||{};
    return response_(p.callback,{ok:false,message:String(err&&err.message||err)});
  }
}

function response_(callback,payload){
  const json=JSON.stringify(payload);
  if(callback){
    const safe=String(callback).replace(/[^\w.$]/g,'');
    return ContentService.createTextOutput(safe+'('+json+');')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(json)
    .setMimeType(ContentService.MimeType.JSON);
}

function rows_(sheetName){
  const sh=db_().getSheetByName(sheetName);
  if(!sh || sh.getLastRow()<2) return [];
  const values=sh.getRange(1,1,sh.getLastRow(),sh.getLastColumn()).getValues();
  const headers=values[0].map(String);
  return values.slice(1).map(r=>{
    const o={};
    headers.forEach((h,i)=>o[h]=r[i]);
    return o;
  });
}

function searchVehicle_(q){
  const query=normalize_(q);
  if(!query) return {found:[]};

  const list=rows_('VEHICLES').map((v,i)=>({
    row:i+2,
    plate:String(v['Nomor Kendaraan']||'').trim(),
    name:String(v['Nama']||'').trim(),
    division:String(v['Bagian/Divisi']||'').trim(),
    type:String(v['Jenis Kendaraan']||'').trim(),
    model:String(v['Model']||'').trim(),
    color:String(v['Warna Kendaraan']||'').trim()
  })).filter(v=>{
    const p=normalize_(v.plate), n=normalize_(v.name);
    return p.indexOf(query)!==-1 || n.indexOf(query)!==-1;
  }).slice(0,50);

  return {found:list};
}

function registerVehicle_(p){
  const row=[
    String(p.name||'').trim(),
    String(p.plate||'').trim(),
    String(p.division||'').trim(),
    String(p.type||'').trim(),
    String(p.model||'').trim(),
    String(p.color||'').trim()
  ];

  if(!row[0] || !row[1])
    return {ok:false,message:'Nama dan Nomor Kendaraan wajib diisi.'};

  const sh=db_().getSheetByName('VEHICLES');
  const last=sh.getLastRow();
  let existing=-1;

  if(last>=2){
    const plates=sh.getRange(2,2,last-1,1).getValues();
    existing=plates.findIndex(r=>normalize_(r[0])===normalize_(row[1]));
  }

  if(existing>=0){
    sh.getRange(existing+2,1,1,6).setValues([row]);
    return {ok:true,mode:'updated',message:'Kendaraan sudah terdaftar. Data diperbarui.',vehicle:row};
  }

  sh.getRange(sh.getLastRow()+1,1,1,6).setValues([row]);
  return {ok:true,mode:'created',message:'Kendaraan berhasil didaftarkan.',vehicle:row};
}

function access_(p){
  const plate=String(p.plate||'').trim();
  const activity=String(p.activity||'').toUpperCase().trim();

  if(!plate) throw Error('Nomor kendaraan kosong.');
  if(activity!=='IN' && activity!=='OUT') throw Error('Aktivitas harus IN atau OUT.');

  const exact=searchVehicle_(plate).found.find(v=>normalize_(v.plate)===normalize_(plate));
  if(!exact) throw Error('KENDARAAN TIDAK TERDAFTAR — AKSES DITOLAK.');

  const lock=LockService.getScriptLock();
  lock.waitLock(15000);

  try{
    const sh=db_().getSheetByName('ACCESS_LOG');
    const n=now_();
    const data=sh.getLastRow()>=2
      ? sh.getRange(2,1,sh.getLastRow()-1,10).getValues()
      : [];

    // Find the latest open IN for this exact vehicle on this date.
    let openRow=-1;
    for(let i=data.length-1;i>=0;i--){
      const r=data[i];
      if(String(r[1])===exact.plate && String(r[8])===n.date && String(r[6]).trim() && !String(r[7]).trim()){
        openRow=i+2;
        break;
      }
    }

    if(activity==='IN'){
      // A second IN while already inside is recorded as a separate event,
      // but the row remains structurally valid.
      const row=[
        exact.name,exact.plate,exact.division,exact.type,exact.model,exact.color,
        n.time,'',n.date,String(p.notes||'')
      ];
      sh.getRange(sh.getLastRow()+1,1,1,10).setValues([row]);
    }else{
      if(openRow>0){
        sh.getRange(openRow,8).setValue(n.time);
        if(String(p.notes||'').trim()) sh.getRange(openRow,10).setValue(String(p.notes));
      }else{
        const row=[
          exact.name,exact.plate,exact.division,exact.type,exact.model,exact.color,
          '',n.time,n.date,String(p.notes||'OUT tanpa IN')
        ];
        sh.getRange(sh.getLastRow()+1,1,1,10).setValues([row]);
      }
    }

    rebuildMerge_();

    return {success:true,vehicle:exact,activity,timestamp:n.timestamp,gate:p.gate||'Gate 1'};
  }finally{
    lock.releaseLock();
  }
}

function rebuildMerge_(){
  const sh=db_().getSheetByName('ACCESS_LOG');
  const last=sh.getLastRow();
  if(last<2) return;

  // Remove previous merges in Name column only.
  sh.getRange(2,1,last-1,1).breakApart();

  // Center name groups for the same person and same date.
  const data=sh.getRange(2,1,last-1,10).getValues();
  let start=2;
  while(start<=last){
    const first=data[start-2];
    const name=String(first[0]||'');
    const date=String(first[8]||'');
    let end=start+1;
    while(end<=last){
      const r=data[end-2];
      if(String(r[0]||'')!==name || String(r[8]||'')!==date) break;
      end++;
    }
    const span=end-start;
    if(span>1){
      sh.getRange(start,1,span,1).mergeVertically();
    }
    sh.getRange(start,1,span,1)
      .setVerticalAlignment('middle')
      .setHorizontalAlignment('center');
    start=end;
  }
}

function dashboard_(){
  const vehicles=rows_('VEHICLES');
  const log=rows_('ACCESS_LOG');
  const today=now_().date;
  const todayRows=log.filter(r=>String(r['Hari/Tanggal']||'')===today);
  return {
    vehicles:vehicles.length,
    owners:new Set(vehicles.map(v=>String(v['Nama']||'').trim()).filter(Boolean)).size,
    inToday:todayRows.filter(r=>String(r['Jam Masuk']||'').trim()).length,
    denied:todayRows.filter(r=>String(r['Keterangan']||'').toUpperCase().indexOf('DITOLAK')>=0).length
  };
}

function current_(){
  const log=rows_('ACCESS_LOG');
  const map={};
  log.forEach(r=>{
    const p=normalize_(r['Nomor Kendaraan']);
    if(!p)return;
    map[p]=r;
  });
  return Object.values(map).filter(r=>String(r['Jam Masuk']||'').trim() && !String(r['Jam Keluar']||'').trim());
}

function report_(p){
  const year=Number(p.year)||now_().year;
  const month=Number(p.month)||now_().month;
  const monthText=String(month).padStart(2,'0');

  const log=rows_('ACCESS_LOG').filter(r=>{
    const d=String(r['Hari/Tanggal']||'');
    return d.slice(6,10)===String(year) && d.slice(3,5)===monthText;
  });

  const grouped=log.map(r=>({
    tanggal:r['Hari/Tanggal']||'',
    nama:r['Nama']||'',
    employeeId:'',
    divisi:r['Bagian/Divisi']||'',
    jabatan:'',
    noPolisi:r['Nomor Kendaraan']||'',
    jenisKendaraan:r['Jenis Kendaraan']||'',
    riwayat:(r['Jam Masuk']?'IN '+r['Jam Masuk']:'')+(r['Jam Keluar']?' → OUT '+r['Jam Keluar']:''),
    jumlahIn:r['Jam Masuk']?1:0,
    jumlahOut:r['Jam Keluar']?1:0,
    firstIn:r['Jam Masuk']||'',
    lastOut:r['Jam Keluar']||'',
    status:r['Jam Keluar']?'SELESAI':'MASIH DI DALAM'
  }));

  return {
    year,month,
    summary:{
      people:new Set(grouped.map(x=>x.nama+'|'+x.tanggal)).size,
      in:grouped.reduce((a,x)=>a+x.jumlahIn,0),
      out:grouped.reduce((a,x)=>a+x.jumlahOut,0),
      totalTransactions:grouped.length
    },
    grouped
  };
}

function health_(){
  const ss=db_();
  return {
    system:'VACS',
    status:'ONLINE',
    version:CONFIG.VERSION,
    timestamp:now_().timestamp,
    spreadsheetId:CONFIG.SPREADSHEET_ID,
    sheets:ss.getSheets().map(s=>s.getName())
  };
}

function diagnostics_(){ return health_(); }

function route_(action,p){
  switch(action){
    case 'health': return health_();
    case 'dashboard': return dashboard_();
    case 'search': return searchVehicle_(p.q||'');
    case 'registerVehicle': return registerVehicle_(p);
    case 'access': return access_(p);
    case 'current': return current_();
    case 'report': return report_(p);
    case 'diagnostics': return diagnostics_();
    default: throw Error('Action tidak dikenal: '+action);
  }
}

/**
 * Manual edits in ACCESS_LOG do not require any other sheet.
 * Keep the Name merge consistent after a manual edit.
 */
function onEdit(e){
  try{
    if(e && e.range && e.range.getSheet().getName()==='ACCESS_LOG'){
      rebuildMerge_();
    }
  }catch(err){}
}
