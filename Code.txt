function doGet(e){
  try{
    ensureDatabase_();
    const p=(e&&e.parameter)||{};
    const data=route_(p.action||'health',p);
    return response_(p.callback,{ok:true,data:data});
  }catch(err){
    const p=(e&&e.parameter)||{};
    return response_(p.callback,{ok:false,message:String(err&&err.message||err)});
  }
}

function response_(callback,payload){
  const json=JSON.stringify(payload);
  if(callback){
    const safe=String(callback).replace(/[^\w.$]/g,'');
    return ContentService.createTextOutput(safe+'('+json+');')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(json)
    .setMimeType(ContentService.MimeType.JSON);
}

function createHeadersOnly_(sh,headers){

function normalize_(v){return String(v||'').toUpperCase().replace(/[^A-Z0-9]/g,'')}

function now_(){

  return {
    timestamp:Utilities.formatDate(d,CONFIG.TIMEZONE,'yyyy-MM-dd HH:mm:ss'),
    date:Utilities.formatDate(d,CONFIG.TIMEZONE,'dd/MM/yyyy'),
    time:Utilities.formatDate(d,CONFIG.TIMEZONE,'HH:mm:ss')
  };
}

function id_(p){
  return p+'-'+Utilities.formatDate(new Date(),CONFIG.TIMEZONE,'yyyyMMddHHmmss')+'-'+Math.floor(Math.random()*100000);
}

function rebuildFinalVehicleLogbook_(ss){
  const sh=ss.getSheetByName('ACCESS_LOG');
  const hist=ss.getSheetByName('ACCESS_LOG');
  if(!sh||!hist)return;

  const H=HEADERS.ACCESS_LOG;
  const data=hist.getDataRange().getValues();
  if(data.length<2){
    sh.getRange(1,1,1,10).setValues([['Nama','Nomor Kendaraan','Bagian/Divisi','Jenis Kendaraan','Model','Warna Kendaraan','Jam Masuk','Jam Keluar','Hari/Tanggal','Keterangan']]);
    return;
  }

  const head=data[0].map(String), ix=n=>head.indexOf(n);
  const vehicles=rows_('VEHICLES');
  const vm={};
  vehicles.forEach(v=>{
    const p=normalize_(v['Nomor Kendaraan']);
    if(p)vm[p]=v;
  });

  const groups={};

    const date=String(r[ix('Tanggal')]||'').trim();
    const plate=String(r[ix('No Polisi')]||'').trim();
    const act=String(r[ix('Activity')]||'').toUpperCase().trim();
    if(!date||!plate||!act)return;

    const v=vm[normalize_(plate)]||{};
    const name=String(v['Nama']||r[ix('Nama Pemilik')]||'Belum diisi').trim();
    const key=date+'|'+normalize_(name||plate);

    if(!groups[key])groups[key]={
      name,
      div:String(v['Bagian/Divisi']||r[ix('Divisi')]||''),
      type:String(v['Jenis Kendaraan']||r[ix('Jenis')]||''),
      model:String(v['Model']||r[ix('Model')]||''),
      color:String(v['Warna Kendaraan']||r[ix('Warna')]||''),
      date,events:[]
    };
    const g=groups[key];
    g.events.push({act,time:String(r[ix('Jam')]||''),plate,note:String(ix('Keterangan')>=0?r[ix('Keterangan')]||'':'')});
  });

  const rows=[];
  Object.values(groups).forEach(g=>{
    g.events.sort((a,b)=>String(a.time).localeCompare(String(b.time)));
    let pending=null;
    g.events.forEach(e=>{
      if(e.act==='IN'){
        if(pending)rows.push([g.name,pending.plate,g.div,g.type,g.model,g.color,pending.time,'',g.date,pending.note||'MASIH DI DALAM']);

          pending=null;
        }else rows.push([g.name,e.plate,g.div,g.type,g.model,g.color,'',e.time,g.date,e.note||'OUT tanpa IN']);

  }

function migrateAccessLogStructure_(ss){

  const logSh=ss.getSheetByName('ACCESS_LOG');
  const histSh=ss.getSheetByName('ACCESS_LOG');
  if(!logSh||!histSh)return;

  const oldHeaders=HEADERS.ACCESS_LOG;
  const groupedHeaders=HEADERS.ACCESS_LOG;
  const logLastRow=logSh.getLastRow();
  const logLastCol=Math.max(1,logSh.getLastColumn());
  const first=logLastRow>0
    ? logSh.getRange(1,1,1,logLastCol).getValues()[0].map(String)
    : [];

  // Old schema = raw transaction columns (Log ID + Activity).
  // Copy ALL missing raw rows into ACCESS_LOG, even when history already
  // contains some rows. This makes the migration idempotent and prevents loss.
  const isOldRaw=first.indexOf('Log ID')>=0 && first.indexOf('Activity')>=0;

  if(isOldRaw && logLastRow>1){
    const raw=logSh.getRange(1,1,logLastRow,Math.min(logLastCol,oldHeaders.length)).getValues();
    const existingIds={};
    const hv=histSh.getDataRange().getValues();
    const hi=hv.length?hv[0].map(String):oldHeaders;
    const idIdx=hi.indexOf('Log ID');
    if(idIdx>=0){
      hv.slice(1).forEach(r=>{
        const id=String(r[idIdx]||'').trim();
        if(id)existingIds[id]=true;
      });
    }

    const body=raw.slice(1).filter(r=>{
      const id=String(r[oldHeaders.indexOf('Log ID')]||'').trim();
      return id && !existingIds[id];
    });

    if(histSh.getLastRow()===0){
      histSh.getRange(1,1,1,oldHeaders.length).setValues([oldHeaders]);
    }
    if(body.length){
      histSh.getRange(histSh.getLastRow()+1,1,body.length,oldHeaders.length).setValues(body);
    }
  }

  // Force ACCESS_LOG to the intended 1x24h operational schema.
  // clearContent changes values only; it does NOT alter width, colors, borders,
  // number formats, frozen rows, filters, or other manual layout settings.
  const maxRows=Math.max(1,logSh.getMaxRows());
  const maxCols=Math.max(groupedHeaders.length,logSh.getMaxColumns());
  logSh.getRange(1,1,maxRows,maxCols).clearContent();
  logSh.getRange(1,1,1,groupedHeaders.length).setValues([groupedHeaders]);

  // Rebuild from the preserved raw history.
    rebuildFinalVehicleLogbook_(ss);
}


function rebuildAccessLog24h_(ss){
  const hist=ss.getSheetByName('ACCESS_LOG');
  const out=ss.getSheetByName('ACCESS_LOG');
  if(!hist||!out)return;

  const hv=hist.getDataRange().getValues();
  if(hv.length<2){
    out.getRange(2,1,Math.max(1,out.getMaxRows()-1),HEADERS.ACCESS_LOG.length).clearContent();
    return;
  }

  const headers=hv[0].map(String);
  const ix=h=>headers.indexOf(h);
  const groups={};

  hv.slice(1).forEach(function(r){
    const date=String(r[ix('Tanggal')]||'').trim();
    const name=String(r[ix('Nama Pemilik')]||'').trim()||'Belum diisi';
    const owner=String(r[ix('Owner ID')]||'').trim();
    const emp=String(r[ix('Employee ID')]||'').trim();
    const plate=String(r[ix('No Polisi')]||'').trim();
    const identity=owner||emp||normalize_(name)||('PLATE-'+normalize_(plate));
    if(!date||!plate)return;
    const key=date+'|'+identity;
    if(!groups[key]){
      groups[key]={date,name,employeeId:emp,division:String(r[ix('Divisi')]||''),position:String(r[ix('Jabatan')]||''),
        plates:{},types:{},events:[],ins:0,outs:0,firstIn:'',lastOut:'',updated:''};
    }
    const g=groups[key];
    if(name&&g.name==='Belum diisi')g.name=name;
    if(emp&&!g.employeeId)g.employeeId=emp;
    if(!g.division)g.division=String(r[ix('Divisi')]||'');
    if(!g.position)g.position=String(r[ix('Jabatan')]||'');
    g.plates[plate]=true;
    const act=String(r[ix('Activity')]||'').toUpperCase();
    const time=String(r[ix('Jam')]||'');
    const ts=String(r[ix('Server Timestamp')]||'');
    if(act==='IN'){g.ins++;if(!g.firstIn)g.firstIn=time}
    if(act==='OUT'){g.outs++;g.lastOut=time}
    g.events.push({act,time,plate,gate:String(r[ix('Gate')]||''),ts});
    if(ts>g.updated)g.updated=ts;
  });

  const vehicleValues=ss.getSheetByName('VEHICLES').getDataRange().getValues();
  const vh=vehicleValues.length?vehicleValues[0].map(String):[];
  const plateType={};
  if(vehicleValues.length>1){
    const pi=vh.indexOf('No Polisi'), ti=vh.indexOf('Jenis');
    vehicleValues.slice(1).forEach(r=>{if(pi>=0&&ti>=0&&r[pi])plateType[normalize_(r[pi])]=String(r[ti]||'')});
  }

  const values=Object.values(groups).sort((a,b)=>(a.date+a.name).localeCompare(b.date+b.name)).map(g=>{
    const plates=Object.keys(g.plates);
    const types=[...new Set(plates.map(p=>plateType[normalize_(p)]).filter(Boolean))];
    const history=g.events.sort((a,b)=>a.ts.localeCompare(b.ts)).map(e=>e.act+' '+e.time+' ['+e.plate+']').join('  |  ');
    const status=g.ins>g.outs?'MASIH DI DALAM':(g.ins===g.outs?'SELESAI':'OUT TANPA IN');
    return [g.date,g.name,g.employeeId,g.division,g.position,plates.join(', '),types.join(', '),history,g.ins,g.outs,g.firstIn,g.lastOut,status,g.updated||now_().timestamp];
  });

  if(out.getMaxRows()>1)out.getRange(2,1,out.getMaxRows()-1,HEADERS.ACCESS_LOG.length).clearContent();
  if(values.length)out.getRange(2,1,values.length,HEADERS.ACCESS_LOG.length).setValues(values);
}

function recordAccess_(p){
  const plate=String(p.plate||'').trim(),act=String(p.activity||'').toUpperCase();
  if(!plate)throw Error('Nomor polisi kosong.');
  if(['IN','OUT'].indexOf(act)<0)throw Error('Activity harus IN atau OUT.');

  const f=searchVehicle_(plate).found;
  const v=f.find(x=>normalize_(x.plate)===normalize_(plate));
  const n=now_(),gate=p.gate||'Gate 1',oid=p.officerId||'SEC-001',on=p.officerName||'Operator',did=p.deviceId||'WEB-DEVICE';

  if(!v){
    append_('ACCESS_LOG',{
      'Denied ID':id_('DEN'),'Server Timestamp':n.timestamp,'Tanggal':n.date,'Jam':n.time,
      'No Polisi':plate.toUpperCase(),'Gate':gate,'Officer ID':oid,'Officer Name':on,
      'Device ID':did,'Reason':'NOT_REGISTERED','Notes':p.notes||''
    },HEADERS.ACCESS_LOG);
    throw Error('KENDARAAN TIDAK TERDAFTAR — AKSES DITOLAK.');
  }

  const lock=LockService.getScriptLock();lock.waitLock(15000);
  try{
    append_('ACCESS_LOG',{
      'Log ID':id_('LOG'),'Server Timestamp':n.timestamp,'Tanggal':n.date,'Jam':n.time,
      'No Polisi':v.plate,'Owner ID':v.ownerId,'Nama Pemilik':v.name,'Employee ID':v.employeeId,
      'Divisi':v.division,'Jabatan':v.position,'Activity':act,'Gate':gate,
      'Officer ID':oid,'Officer Name':on,'Device ID':did,'Notes':p.notes||''
    },HEADERS.ACCESS_LOG);
    rebuildAccessLog24h_(db_());
  }finally{lock.releaseLock()}

  // Keep ACCESS_LOG synchronized automatically after every real-time transaction.
  // This does not touch ACCESS_LOG history and does not resize/reformat the sheet.
  try{
    report24h_({
      year:Number(String(n.date).split('/')[2]),
      month:Number(String(n.date).split('/')[1]),
      day:Number(String(n.date).split('/')[0]),
      gate:'ALL',
      activity:'ALL'
    });
  }catch(reportErr){
    // Transaction must remain successful even if report-view refresh fails.
    console.log('ACCESS_LOG refresh skipped: '+reportErr.message);
  }

  try{rebuildFinalVehicleLogbook_(database_());}catch(e){console.log('logbook rebuild: '+e.message)}
  return{success:true,vehicle:v,activity:act,timestamp:n.timestamp,gate:gate};
}

function csvParse_(text){
  const rows=[],row=[];
  let cell='',quote=false;

  for(let i=0;i<text.length;i++){
    const c=text[i],next=text[i+1];

    if(c==='"'){
      if(quote&&next==='"'){cell+='"';i++}
      else quote=!quote;
    }else if(c===','&&!quote){
      row.push(cell);cell='';
    }else if((c==='\n'||c==='\r')&&!quote){
      if(c==='\r'&&next==='\n')i++;
      row.push(cell);cell='';
      if(row.some(x=>String(x).trim()!==''))rows.push(row.splice(0,row.length));
    }else{
      cell+=c;
    }
  }
  if(cell!==''||row.length){row.push(cell);if(row.some(x=>String(x).trim()!==''))rows.push(row)}
  return rows;
}

function cleanHeader_(s){
  return String(s||'').trim().toLowerCase()
    .replace(/[._-]/g,' ')
    .replace(/\s+/g,' ');
}

function findHeader_(headers,names){
  const normalized=headers.map(cleanHeader_);
  for(const n of names){
    const idx=normalized.indexOf(cleanHeader_(n));
    if(idx>=0)return idx;
  }
  return -1;
}

  }

  let ownerInserted=0,ownerUpdated=0,vehicleInserted=0,vehicleUpdated=0,skipped=0,duplicateSameOwner=0,ownershipConflicts=0;
  const updatesOwners=[],updatesVehicles=[];

  matrix.slice(1).forEach(function(r,idx){
    const plate=String(r[iPlate]||'').trim();
    const name=iName>=0?String(r[iName]||'').trim():'';

    const emp=iEmp>=0?String(r[iEmp]||'').trim():'';
    const ownerKey=normalize_(emp||name||('PLATE-'+normalize_(plate)));
    let owner=ownerByKey[ownerKey];

        row:null,

      };

      const newRow=new Array(oh.length).fill('');
      newRow[ownerIdCol]=ownerId;
      newRow[ownerNameCol]=name;
      if(ownerEmpCol>=0)newRow[ownerEmpCol]=emp;
      const put=(h,names,val)=>{
        const i=findHeader_(oh,names);
        if(i>=0)newRow[i]=val;
      };
      put(oh,['Divisi','Division'],iDiv>=0?r[iDiv]:'');
      put(oh,['Jabatan','Position'],iPos>=0?r[iPos]:'');
      put(oh,['No HP','HP','Phone'],iPhone>=0?r[iPhone]:'');
      put(oh,['Email'],iEmail>=0?r[iEmail]:'');
      put(oh,['Status'],'ACTIVE');
      put(oh,['Updated At'],now_().timestamp);

      ownerSh.appendRow(newRow);
      owner.row=ownerSh.getLastRow();
      ownerByKey[ownerKey]=owner;
      ownerInserted++;
    }else{
      const rowNum=owner.row;
      const current=ownerSh.getRange(rowNum,1,1,oh.length).getValues()[0];
      const set=(names,val)=>{
        const i=findHeader_(oh,names);
        if(i>=0&&val!==undefined&&val!=='')current[i]=val;
      };
      set(['Nama'],name);
      set(['Employee ID'],emp);
      set(['Divisi','Division'],iDiv>=0?r[iDiv]:'');
      set(['Jabatan','Position'],iPos>=0?r[iPos]:'');
      set(['No HP','HP','Phone'],iPhone>=0?r[iPhone]:'');
      set(['Email'],iEmail>=0?r[iEmail]:'');
      set(['Status'],'ACTIVE');
      set(['Updated At'],now_().timestamp);
      updatesOwners.push({row:rowNum,values:current});
      ownerUpdated++;
    }

    const plateKey=normalize_(plate);
    const existing=vehicleByPlate[plateKey];

    if(existing){
      const current=vehicleSh.getRange(existing.row,1,1,vh.length).getValues()[0];
      const existingOwnerId=String(current[vh.indexOf('Owner ID')]||'');
      if(existingOwnerId && owner.id && existingOwnerId !== String(owner.id)){
        ownershipConflicts++;
        skipped++;
        return;
      }
      duplicateSameOwner++;
      const set=(names,val)=>{
        const i=findHeader_(vh,names);
        if(i>=0&&val!==undefined&&val!=='')current[i]=val;
      };
      set(['Owner ID'],owner.id);
      set(['Merk','Brand'],iBrand>=0?r[iBrand]:'');
      set(['Model','Tipe'],iModel>=0?r[iModel]:'');
      set(['Tahun','Year'],iYear>=0?r[iYear]:'');
      set(['Warna','Color'],iColor>=0?r[iColor]:'');
      set(['Jenis','Type'],iType>=0?r[iType]:'');
      set(['Status'],'ACTIVE');
      set(['Updated At'],now_().timestamp);
      updatesVehicles.push({row:existing.row,values:current});
      vehicleUpdated++;
    }else{
      const newRow=new Array(vh.length).fill('');
      newRow[vehicleIdCol]=id_('VH');
      newRow[vehiclePlateCol]=plate.toUpperCase();
      const set=(names,val)=>{
        const i=findHeader_(vh,names);
        if(i>=0)newRow[i]=val;
      };
      set(['Owner ID'],owner.id);
      set(['Merk','Brand'],iBrand>=0?r[iBrand]:'');
      set(['Model','Tipe'],iModel>=0?r[iModel]:'');
      set(['Tahun','Year'],iYear>=0?r[iYear]:'');
      set(['Warna','Color'],iColor>=0?r[iColor]:'');
      set(['Jenis','Type'],iType>=0?r[iType]:'');
      set(['Status'],'ACTIVE');
      set(['Updated At'],now_().timestamp);
      vehicleSh.appendRow(newRow);
      vehicleByPlate[plateKey]={row:vehicleSh.getLastRow()};
      vehicleInserted++;
    }
  });

  updatesOwners.forEach(x=>ownerSh.getRange(x.row,1,1,oh.length).setValues([x.values]));
  updatesVehicles.forEach(x=>vehicleSh.getRange(x.row,1,1,vh.length).setValues([x.values]));

  return {
    success:true,
    ownersInserted:ownerInserted,
    ownersUpdated:ownerUpdated,
    vehiclesInserted:vehicleInserted,
    vehiclesUpdated:vehicleUpdated,
    duplicateSameOwner:duplicateSameOwner,
    ownershipConflicts:ownershipConflicts,
    skipped:skipped
  };
}

function saveMasterManual_(p){
  const plate=String(p.plate||'').trim();
  if(!plate) throw Error('No Polisi wajib diisi.');

  const ownerSh=sheet_('VEHICLES'), vehicleSh=sheet_('VEHICLES');
  const oh=ownerSh.getDataRange().getValues()[0];
  const vh=vehicleSh.getDataRange().getValues()[0];
  const ov=ownerSh.getDataRange().getValues();
  const vv=vehicleSh.getDataRange().getValues();

  const plateKey=normalize_(plate);
  let ownerId='';
  const emp=String(p.employeeId||'').trim();
  const name=String(p.name||'').trim();

  // Find owner by Employee ID first, then Name.
  for(let r=1;r<ov.length;r++){
    const o={};oh.forEach((h,i)=>o[h]=ov[r][i]);
    if((emp && normalize_(o['Employee ID'])===normalize_(emp)) ||
       (!emp && name && normalize_(o['Nama'])===normalize_(name))){
      ownerId=o['Owner ID']; break;
    }
  }

  // Existing vehicle by plate.
  let vehicleRow=-1;
  for(let r=1;r<vv.length;r++){
    if(normalize_(vv[r][vh.indexOf('No Polisi')])===plateKey){vehicleRow=r+1;ownerId=vv[r][vh.indexOf('Owner ID')]||ownerId;break;}
  }

  if(!ownerId){
    ownerId=id_('OWN');
    const row=new Array(oh.length).fill('');
    const put=(h,v)=>{const i=oh.indexOf(h);if(i>=0)row[i]=v};
    put('Owner ID',ownerId); put('Nama',name); put('Employee ID',emp);
    put('Divisi',p.division||''); put('Jabatan',p.position||'');
    put('No HP',p.phone||''); put('Email',p.email||''); put('Status','ACTIVE');
    put('Catatan',p.ownerNotes||''); put('Updated At',now_().timestamp);
    ownerSh.appendRow(row);
  }else{
    // Fill only fields supplied: later enrichment never wipes existing profile.
    const rowNum=(function(){
      for(let r=1;r<ov.length;r++){
        if(String(ov[r][oh.indexOf('Owner ID')])===String(ownerId)) return r+1;
      }
      return -1;
    })();
    if(rowNum>1){
      const row=ownerSh.getRange(rowNum,1,1,oh.length).getValues()[0];
      const set=(h,v)=>{const i=oh.indexOf(h);if(i>=0&&String(v??'').trim()!=='')row[i]=v};
      set('Nama',name);set('Employee ID',emp);set('Divisi',p.division);
      set('Jabatan',p.position);set('No HP',p.phone);set('Email',p.email);
      set('Catatan',p.ownerNotes);set('Status','ACTIVE');set('Updated At',now_().timestamp);
      ownerSh.getRange(rowNum,1,1,oh.length).setValues([row]);
    }
  }

  if(vehicleRow<0){
    const row=new Array(vh.length).fill('');
    const put=(h,v)=>{const i=vh.indexOf(h);if(i>=0)row[i]=v};
    put('Vehicle ID',id_('VH'));put('No Polisi',plate.toUpperCase());put('Owner ID',ownerId);
    put('Merk',p.brand||'');put('Model',p.model||'');put('Tahun',p.year||'');
    put('Warna',p.color||'');put('Jenis',p.type||'');put('Status','ACTIVE');
    put('Catatan',p.vehicleNotes||'');put('Updated At',now_().timestamp);
    vehicleSh.appendRow(row);
    return {success:true,mode:'CREATED',plate:plate.toUpperCase(),ownerId:ownerId};
  }else{
    const row=vehicleSh.getRange(vehicleRow,1,1,vh.length).getValues()[0];
    const set=(h,v)=>{const i=vh.indexOf(h);if(i>=0&&String(v??'').trim()!=='')row[i]=v};
    set('Owner ID',ownerId);set('Merk',p.brand);set('Model',p.model);set('Tahun',p.year);
    set('Warna',p.color);set('Jenis',p.type);set('Catatan',p.vehicleNotes);set('Status','ACTIVE');
    set('Updated At',now_().timestamp);
    vehicleSh.getRange(vehicleRow,1,1,vh.length).setValues([row]);
    return {success:true,mode:'UPDATED',plate:plate.toUpperCase(),ownerId:ownerId};
  }
}

function parseDate_(s){
  const p=String(s||'').split('-');
  if(p.length!==3)return null;
  return new Date(Number(p[0]),Number(p[1])-1,Number(p[2]));
}

function report24h_(p){
  const year=Number(p.year||new Date().getFullYear());
  const month=String(p.month||'ALL').toUpperCase();
  const day=String(p.day||'ALL').toUpperCase();
  const gate=String(p.gate||'ALL').toUpperCase();
  const activity=String(p.activity||'ALL').toUpperCase();

  const vehicles=rows_('VEHICLES');
  const vehicleMap={};
  vehicles.forEach(function(v){
    vehicleMap[normalize_(v['No Polisi'])]={
      type:String(v['Jenis']||''),
      ownerId:String(v['Owner ID']||'')
    };
  });

  const logs=rows_('ACCESS_LOG').filter(function(x){
    const parts=String(x['Tanggal']||'').split('/');
    if(parts.length!==3)return false;
    const d=Number(parts[0]),m=Number(parts[1]),y=Number(parts[2]);
    if(y!==year)return false;
    if(month!=='ALL'&&m!==Number(month))return false;
    if(day!=='ALL'&&d!==Number(day))return false;
    if(gate!=='ALL'&&String(x['Gate']||'').toUpperCase()!==gate)return false;
    return true;
  }).sort(function(a,b){
    return String(a['Server Timestamp']).localeCompare(String(b['Server Timestamp']));
  });

  const displayLogs=activity==='ALL'
    ? logs
    : logs.filter(function(x){return String(x['Activity']||'').toUpperCase()===activity;});

  // One row = one owner/person per calendar day (the operational 1x24h window).
  // Owner ID is preferred, then Employee ID, then name, then plate as fallback.
  const groups={};
  logs.forEach(function(x){
    const date=String(x['Tanggal']||'');
    const ownerId=String(x['Owner ID']||'').trim();
    const name=String(x['Nama Pemilik']||'Belum diisi').trim()||'Belum diisi';
    const emp=String(x['Employee ID']||'').trim();
    const identity=ownerId||emp||normalize_(name)||('PLATE-'+normalize_(x['No Polisi']));
    const key=date+'|'+identity;

    if(!groups[key]){
      groups[key]={
        date:date,name:name,employeeId:emp,ownerId:ownerId,
        division:String(x['Divisi']||''),position:String(x['Jabatan']||''),
        plates:{},types:{},events:[],ins:0,outs:0,firstIn:'',lastOut:''
      };
    }

    const g=groups[key];
    if(!g.name||g.name==='Belum diisi')g.name=name;
    if(!g.employeeId)g.employeeId=emp;
    if(!g.ownerId)g.ownerId=ownerId;
    if(!g.division)g.division=String(x['Divisi']||'');
    if(!g.position)g.position=String(x['Jabatan']||'');

    const plate=String(x['No Polisi']||'').trim();
    const v=vehicleMap[normalize_(plate)];
    if(plate)g.plates[plate]=true;
    if(v&&v.type)g.types[v.type]=true;

    const act=String(x['Activity']||'').toUpperCase();
    const time=String(x['Jam']||'');
    if(act==='IN'){g.ins++;if(!g.firstIn)g.firstIn=time;}
    if(act==='OUT'){g.outs++;g.lastOut=time;}
    g.events.push({act:act,time:time,plate:plate});
  });

  const grouped=Object.keys(groups).map(function(k){
    const g=groups[k];
    const timeline=g.events.map(function(e){
      return e.act+' '+e.time+(e.plate?' ['+e.plate+']':'');
    }).join('  |  ');
    const plates=Object.keys(g.plates).join(', ');
    const types=Object.keys(g.types).join(', ');
    let status='SELESAI';
    if(g.ins>g.outs)status='MASIH DI DALAM';
    if(g.ins===0&&g.outs>0)status='OUT TANPA IN';

    // Activity filter only controls displayed rows/timeline; grouping remains
    // based on the complete 1x24h day so the person's status is not distorted.
    const visibleEvents=activity==='ALL'
      ? g.events
      : g.events.filter(function(e){return e.act===activity;});
    const visibleTimeline=visibleEvents.map(function(e){
      return e.act+' '+e.time+(e.plate?' ['+e.plate+']':'');
    }).join('  |  ');

    return {
      tanggal:g.date,nama:g.name,employeeId:g.employeeId,divisi:g.division,
      jabatan:g.position,noPolisi:plates,jenisKendaraan:types,
      riwayat:visibleTimeline||timeline,jumlahIn:g.ins,jumlahOut:g.outs,
      firstIn:g.firstIn,lastOut:g.lastOut,status:status,ownerId:g.ownerId
    };
  }).filter(function(g){
    return activity==='ALL'||g.riwayat;
  }).sort(function(a,b){
    return (a.tanggal+a.nama).localeCompare(b.tanggal+b.nama);
  });

  // ACCESS_LOG is a generated data view. Preserve the sheet's manually
  // configured widths, colors, fonts, freeze panes, etc.
  const sh=sheet_('ACCESS_LOG');
  const headers=HEADERS.ACCESS_LOG;
  if(sh.getLastRow()===0){
    sh.getRange(1,1,1,headers.length).setValues([headers]);
  }else{
    const existing=sh.getRange(1,1,1,headers.length).getValues()[0];
    if(existing.every(function(v){return String(v||'').trim()==='';})){
      sh.getRange(1,1,1,headers.length).setValues([headers]);
    }
  }

  if(sh.getLastRow()>1){
    sh.getRange(2,1,sh.getLastRow()-1,headers.length).clearContent();
  }

  const values=grouped.map(function(g){
    return [
      g.tanggal,g.nama,g.employeeId,g.divisi,g.jabatan,g.noPolisi,
      g.jenisKendaraan,g.riwayat,g.jumlahIn,g.jumlahOut,g.firstIn,g.lastOut,
      g.status,now_().timestamp
    ];
  });
  if(values.length)sh.getRange(2,1,values.length,headers.length).setValues(values);

  // ACCESS_LOG is the live 1x24h operational view.
  rebuildAccessLog24h_(db_());

  return {
    filters:{year:year,month:month,day:day,gate:gate,activity:activity},
    summary:{
      totalTransactions:logs.length,
      visibleTransactions:displayLogs.length,
      people:grouped.length,
      in:logs.filter(function(x){return String(x['Activity']).toUpperCase()==='IN';}).length,
      out:logs.filter(function(x){return String(x['Activity']).toUpperCase()==='OUT';}).length
    },
    grouped:grouped,
    logs:displayLogs
  };
}

function report_(p){
  const year=Number(p.year||new Date().getFullYear());
  const month=String(p.month||'ALL').toUpperCase();
  const day=String(p.day||'ALL').toUpperCase();
  const gate=String(p.gate||'ALL').toUpperCase();
  const activity=String(p.activity||'ALL').toUpperCase();

  const logs=rows_('ACCESS_LOG').filter(function(x){
    const raw=String(x['Tanggal']||'');
    const parts=raw.split('/');
    if(parts.length!==3)return false;

    const d=Number(parts[0]),m=Number(parts[1]),y=Number(parts[2]);

    if(y!==year)return false;
    if(month!=='ALL'&&m!==Number(month))return false;
    if(day!=='ALL'&&d!==Number(day))return false;
    if(gate!=='ALL'&&String(x['Gate']||'').toUpperCase()!==gate)return false;
    if(activity!=='ALL'&&String(x['Activity']||'').toUpperCase()!==activity)return false;

    return true;
  });

  const denied=rows_('ACCESS_LOG').filter(function(x){
    const parts=String(x['Tanggal']||'').split('/');
    if(parts.length!==3)return false;
    const d=Number(parts[0]),m=Number(parts[1]),y=Number(parts[2]);

    if(y!==year)return false;
    if(month!=='ALL'&&m!==Number(month))return false;
    if(day!=='ALL'&&d!==Number(day))return false;
    if(gate!=='ALL'&&String(x['Gate']||'').toUpperCase()!==gate)return false;

    return true;
  });

  return {
    filters:{year:year,month:month,day:day,gate:gate,activity:activity},
    summary:{
      total:logs.length,
      in:logs.filter(x=>String(x['Activity']).toUpperCase()==='IN').length,
      out:logs.filter(x=>String(x['Activity']).toUpperCase()==='OUT').length,
      denied:denied.length
    },
    logs:logs,
    deniedLogs:denied
  };
}

function currentStatus_(){
  const logs=rows_('ACCESS_LOG');
  const latest={};

  logs.forEach(function(x){
    const plate=normalize_(x['No Polisi']);
    if(!plate)return;

    if(!latest[plate] ||
       String(x['Server Timestamp'])>String(latest[plate]['Server Timestamp'])){
      latest[plate]=x;
    }
  });

  return Object.keys(latest).map(function(k){
    const x=latest[k];
    return {
      plate:x['No Polisi'],
      ownerId:x['Owner ID'],
      name:x['Nama Pemilik'],
      division:x['Divisi'],
      position:x['Jabatan'],
      activity:x['Activity'],
      timestamp:x['Server Timestamp'],
      gate:x['Gate']
    };
  }).filter(x=>String(x.activity).toUpperCase()==='IN');
}

function route_(action,p){
  switch(action){
    case 'health':
      return {
        system:'VACS',
        status:'ONLINE',
        timestamp:now_().timestamp,
        spreadsheetId:CONFIG.SPREADSHEET_ID,
        database:'READY',
        layoutPolicy:'PRESERVE_MANUAL',centralDatabase:'SINGLE_SOURCE_OF_TRUTH',duplicatePolicy:'PLATE_UNIQUE'
      };

    case 'search':return searchVehicle_(p.q||'');
    case 'access':return recordAccess_(p);
    case 'dashboard':{
      const d=report_({year:new Date().getFullYear(),month:new Date().getMonth()+1,day:new Date().getDate()});
      return {
        today:now_().date,
        owners:rows_('VEHICLES').length,
        vehicles:rows_('VEHICLES').length,
        totalToday:d.summary.total,
        inToday:d.summary.in,
        outToday:d.summary.out,
        deniedToday:d.summary.denied
      };
    }
    case 'recent':
      return rows_('ACCESS_LOG').reverse().slice(0,Number(p.limit||10)).map(x=>({
        timestamp:x['Server Timestamp'],plate:x['No Polisi'],name:x['Nama Pemilik'],
        activity:x['Activity'],gate:x['Gate'],officerName:x['Officer Name'],deviceId:x['Device ID']
      }));
    case 'setup':return setupVACS();
    case 'report':return report24h_(p);
    case 'reportRaw':return report_(p);
    case 'current':return currentStatus_();
    case 'diagnostics':return diagnostics_();
    case 'registerVehicle':return registerVehicle_(p);
    case 'migrateAccessLog':ensureDatabase_();return {success:true,message:'ACCESS_LOG 1x24h rebuilt.',version:'6.3'};
    case 'importCSV':return importMasterCSV_(p.csv||'');
    case 'saveMaster':return saveMasterManual_(p);
    default:throw Error('Action API tidak dikenal: '+action);
  }
}


function registerVehicle_(p){
  const name=String(p.name||'').trim();
  const plate=String(p.plate||'').trim();
  const division=String(p.division||'').trim();
  const type=String(p.type||'').trim();
  const model=String(p.model||'').trim();
  const color=String(p.color||'').trim();
  if(!name||!plate)return {ok:false,message:'Nama dan Nomor Kendaraan wajib diisi.'};

  const sh=database_().getSheetByName('VEHICLES');
  const headers=['Nama','Nomor Kendaraan','Bagian/Divisi','Jenis Kendaraan','Model','Warna Kendaraan'];
  if(sh.getLastRow()===0)sh.getRange(1,1,1,headers.length).setValues([headers]);

  const data=sh.getDataRange().getValues();
  const hp=data[0].map(String);
  const pi=hp.indexOf('Nomor Kendaraan');
  const existing=data.slice(1).findIndex(r=>normalize_(r[pi])===normalize_(plate));

  const row=[name,plate,division,type,model,color];
  if(existing>=0){
    sh.getRange(existing+2,1,1,headers.length).setValues([row]);
    return {ok:true,mode:'updated',message:'Kendaraan sudah terdaftar. Data diperbarui.',vehicle:row};
  }
  sh.appendRow(row);
  return {ok:true,mode:'created',message:'Kendaraan berhasil didaftarkan.',vehicle:row};
}

/**
 * Optional spreadsheet-side sync:
 * If an administrator manually edits ACCESS_LOG, rebuild today's ACCESS_LOG.
 * This never changes column widths or formatting.
 */
function onEdit(e){
  try{
    if(!e || !e.range) return;
    const sh=e.range.getSheet();
    if(sh.getName()!=='ACCESS_LOG') return;

  }catch(err){
    console.log('onEdit ACCESS_LOG sync: '+err.message);
  }
}
